accumulate_lits <- function(lits, clause_size) {
    mclause <- length(lits) %/% clause_size

    index <- 0
    for (m in 1:mclause) {
        terms <- rep(NA, times = clause_size)
        for (i in 1:clause_size) {
            index <- index + 1
            terms[i] <- lits[index]
        }
        ## construct or_clause
        clause <-or_clause(terms[[1]], terms[[2]], terms[[3]])
        if (m == 1){
            form <- clause
        } else {
            form <- and_clause(form, clause)
        }
    }
    form
}

}

rename_succ <- function(x){
    retvec <- 0 * x

monte <- function(){
    set.seed(42)
    n <- 30
    init <- tibble(id = 1:n) |>
        mutate(string = rpois(n, 5)) |>
        group_by(string) |>
        mutate(rename = min(id)) |>
        arrange(rename) |>
        ungroup() |>
        mutate(
            sign = rrademacher(n, 8/9),
            rename = rename_succ(rename),
            nlit = max(rename))

    or_frame <- mutate(init,
                       group_clause = rep(1:(n/3), each = 3)) |>
        group_by(group_clause) |>
        summarize(or_clause = list(or_vec(rename * sign )))

    wac <- reduce(or_frame$or_clause, and_clause)

    nlit <- init$nlit[1]
    witframe <- tibble(rename = 1:nlit, assignment = rbinom(nlit, 1, 8/9))

    evaluator_frame <- left_join(init, witframe)


    evaluator <- function(assignment, sign){
        evaluation <- if_else(sign ==1,
                              assignment,
                              1 - assignment)
    }

    valuation <- function(evaluator_frame) {
        with(evaluator_frame,
             assignment,  sign)
    }

    satisfaction <- function(valuation) {
        groups <- rep(1:(length(valuation)/3), each = 3)

        reduce(lapply(split(valuation, groups), sum), prod)
    }

    satisfaction(valuation(evaluator_frame))
}
